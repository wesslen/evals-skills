    ```python
    import sys

    import pandas as pd

    BATCH = sys.argv[1] if len(sys.argv) > 1 else "fixtures/batches/batch_13.csv"
    COLUMNS = ["complaint_id", "product", "issue", "udaap_flag", "fair_lending_flag", "evidence"]

    LABELS = {
        'C-1301': ('Credit card', 'Credit limit change', 'N', 'N', ''),
        'C-1302': ('Checking/Savings', 'Transfers and automatic payments', 'N', 'N', ''),
        'C-1303': ('Mortgage', 'Loan modification or forbearance', 'N', 'N', ''),
        'C-1304': ('Debt collection', 'Validation or verification', 'N', 'N', ''),
        'C-1305': ('Personal loan', 'Payment problem', 'N', 'N', ''),
    }

    df = pd.read_csv(BATCH)
    out = pd.DataFrame([(cid, *LABELS[cid]) for cid in df["complaint_id"]], columns=COLUMNS)
    print(f"Triaged {len(out)} of {len(df)} complaints")
    out.to_csv("outputs/triage_output.csv", index=False)
    print("Wrote outputs/triage_output.csv")
    ```
