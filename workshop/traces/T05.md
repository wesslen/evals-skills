    ```bash
    import pandas as pd
    df = pd.read_csv('fixtures/batches/batch_05.csv')
    labels = {
        'C-0501': ('Auto loan', 'Add-on products or GAP coverage', 'high'),
        'C-0502': ('Credit card', 'Billing dispute', 'low'),
        'C-0503': ('Checking/Savings', 'Transfers and automatic payments', 'low'),
        'C-0504': ('Debt collection', 'Credit reporting of collected debt', 'low'),
        'C-0505': ('Mortgage', 'Late fees', 'medium'),
    }
    df[['product', 'issue', 'reg_risk']] = df['complaint_id'].map(labels).apply(pd.Series)
    out = df[['complaint_id', 'product', 'issue', 'reg_risk']]
    out.to_csv('fixtures/batches/batch_05_labeled.csv', index=False)
    print(out.to_string(index=False))
    EOF
    ```
